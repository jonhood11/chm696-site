# Monte Carlo and Markov processes with Python

## Monte Carlo estimation

```{code-cell} python
import numpy as np
import matplotlib.pyplot as plt

rng = np.random.default_rng(696)
points = rng.uniform(-1, 1, size=(60000, 2))
inside = np.sum(points**2, axis=1) <= 1
running_pi = 4*np.cumsum(inside) / np.arange(1, inside.size + 1)

fig, axes = plt.subplots(1, 2, figsize=(10, 4))
axes[0].scatter(points[:2500, 0], points[:2500, 1], c=inside[:2500], s=4, cmap="coolwarm")
axes[0].set(aspect="equal", xlabel="$x$", ylabel="$y$", title="Samples inside and outside the circle")
axes[1].semilogx(np.arange(1, inside.size + 1), running_pi, label="estimate")
axes[1].axhline(np.pi, color="k", linestyle="--", label="$\\pi$")
axes[1].set(xlabel="number of samples", ylabel="estimate", title="Monte Carlo convergence")
axes[1].legend(frameon=False)
fig.tight_layout()
plt.show()
```

## Relaxation of a Markov chain

```{code-cell} python
transition = np.array([[0.88, 0.22], [0.12, 0.78]])
state = np.array([1.0, 0.0])
history = [state.copy()]
for _ in range(35):
    state = transition @ state
    history.append(state.copy())
history = np.array(history)

fig, ax = plt.subplots(figsize=(7, 4))
ax.plot(history[:, 0], "o-", label="state A")
ax.plot(history[:, 1], "o-", label="state B")
ax.set(xlabel="step", ylabel="probability", title="Approach to the stationary distribution")
ax.legend(frameon=False)
ax.grid(alpha=0.2)
plt.show()
```
